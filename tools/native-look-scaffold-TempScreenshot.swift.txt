import Foundation
import SwiftUI

enum TempScreenshot {
    private static let d = UserDefaults.standard

    /// Suppresses the notification permission alert, which otherwise covers the
    /// middle of every shot 5s after launch and survives app termination.
    static var quiet: Bool { d.bool(forKey: "temp_screenshot_quiet") }

    static var tab: AppTab? {
        switch d.string(forKey: "temp_screenshot_tab") {
        case "discover": return .discover
        case "sports": return .feed
        case "browse": return .leagues
        case "search": return .search
        case "mystuff": return .myStuff
        default: return nil
        }
    }

    static var hubSlug: String? { d.string(forKey: "temp_screenshot_hub") }
    static var hubName: String { d.string(forKey: "temp_screenshot_hub_name") ?? "US Open 2026" }
    static var eventID: Int? {
        let v = d.integer(forKey: "temp_screenshot_event")
        return v > 0 ? v : nil
    }

    /// 0 = top, 1 = bottom.
    static var scrollFraction: CGFloat? {
        guard d.object(forKey: "temp_screenshot_scroll") != nil else { return nil }
        return CGFloat(d.double(forKey: "temp_screenshot_scroll"))
    }

    /// Published from `Bain_LuckApp`'s `.task`, after a delay, so the observers
    /// are mounted and see a real change. Route wins over a bare tab.
    @MainActor
    static func seed(_ nav: NavigationCoordinator) async {
        guard tab != nil || hubSlug != nil || eventID != nil else { return }
        try? await Task.sleep(nanoseconds: 2_500_000_000)
        if let slug = hubSlug {
            nav.navigate(to: .tournamentHub(slug: slug, name: hubName), tab: .feed)
        } else if let id = eventID {
            nav.navigate(to: .eventDetail(id: id), tab: .feed)
        } else if let tab {
            nav.selectedTab = tab
        }
    }
}
